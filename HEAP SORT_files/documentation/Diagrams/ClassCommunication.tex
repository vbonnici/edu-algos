\documentclass[tikz,border=10pt]{standalone}
\usetikzlibrary{arrows.meta,positioning,shadows,shapes.geometric}

\begin{document}
\begin{tikzpicture}[font=\sffamily, thick,
    object/.style={draw, fill=white, inner sep=10pt, drop shadow, rectangle, minimum height=1cm, minimum width=2cm},
    actor_head/.style={circle, fill=white, draw=black, minimum size=0.5cm},
    lifeline/.style={draw, dashed},
    activation/.style={draw, fill=white, minimum width=10pt},
    mess/.style={-{Stealth[scale=1.2]}, thick}
]

% Stickman drawing command
\newcommand{\stickman}[2]{
    \node[actor_head] (#1_head) at (#2) {};
    \draw[thick] (#1_head.south) -- ++(0,-0.8) coordinate (#1_body);
    \draw[thick] (#1_body) -- ++(-0.5,-0.8); % Left leg
    \draw[thick] (#1_body) -- ++(0.5,-0.8);  % Right leg
    \draw[thick] (#1_head.south)++(0,-0.3) -- ++(-0.5,0); % Left arm
    \draw[thick] (#1_head.south)++(0,-0.3) -- ++(0.5,0);  % Right arm
    \node[below=1.8cm of #1_head] (#1) {Computer};
}

% Actors
\node[object] (Presenter) {GamePresenter};

% Place Computer (Stickman)
\coordinate (ComputerPos) at (3.5, 0.5); 
\stickman{Computer}{ComputerPos}

\node[object, right=3.5cm of Computer] (Model) {Heap};
\node[object, right=3cm of Model] (View) {Renderer};

% Lifelines
\draw[lifeline] (Presenter) -- ++(0,-9) coordinate (PresenterBottom);
\draw[lifeline] (Computer) -- ++(0,-7.5) coordinate (ComputerBottom); % Adjust length relative to stickman label
\draw[lifeline] (Model) -- ++(0,-9) coordinate (ModelBottom);
\draw[lifeline] (View) -- ++(0,-9) coordinate (ViewBottom);

% Time steps
\coordinate (t1) at (0,-1.5);
\coordinate (t2) at (0,-2.5);
\coordinate (t3) at (0,-3.5);
\coordinate (t4) at (0,-4.5);
\coordinate (t5) at (0,-5.5);
\coordinate (t6) at (0,-6.5);
\coordinate (t7) at (0,-7.5);

% 1. computerTurn()
\draw[mess] (Presenter |- t1) -- ++(-1.5,0) node[left] {checkAutoExtraction()} -- (Presenter |- t1);
\node[activation, minimum height=6.5cm] (PresAct) at (Presenter |- t4.5) {};

% 2. makeMove()
\draw[mess] (Presenter |- t2) -- node[above,midway] {\scriptsize 1. makeMove()} (Computer |- t2);
\node[activation, minimum height=3cm] (CompAct) at (Computer |- t3.5) {};

% 3. isRootValid() / check strategy
\draw[mess] (Computer |- t3) -- node[above,midway] {\scriptsize 2. scanStrategy()} (Model |- t3);
\node[activation, minimum height=1cm] at (Model |- t3.5) {};

% 4. swap()
\draw[mess] (Computer |- t4) -- node[above,midway] {\scriptsize 3. swap(i, j)} (Model |- t4);
\node[activation, minimum height=0.5cm] at (Model |- t4) {};

% 5. Return action
\draw[mess, dashed] (Computer |- t5) -- node[above,midway] {\scriptsize return 'swap'} (Presenter |- t5);

% 6. Update UI
\draw[mess] (Presenter |- t6) -- node[above,midway] {\scriptsize 4. render(heap)} (View |- t6);
\node[activation, minimum height=0.5cm] at (View |- t6) {};

% 7. Check Win
\draw[mess] (Presenter |- t7) -- node[above,midway] {\scriptsize 5. checkWinCondition()} (Presenter |- t7);

\end{tikzpicture}
